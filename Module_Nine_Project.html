# Prepare population data for county-level analysis
# Keep county records only and aggregate demographic groups into
# one population total for each county and year.
population_clean = population_data[
    population_data["county_indicator"] == 1
].copy()

population_county = (
    population_clean
    .groupby(["county_lhj", "year"], as_index=False)["population"]
    .sum()
)

# Prepare unemployment data for county-level annual analysis
# Keep annual county records so the geographic and time levels
# match the population dataset.
unemployment_clean = unemployment_data[
    (unemployment_data["Area Type"] == "County") &
    (unemployment_data["Month"] == "Annual")
].copy()

# Standardize county names and column names before blending.
unemployment_clean["county_lhj"] = (
    unemployment_clean["Area Name"]
    .str.replace(" County", "", regex=False)
    .str.strip()
)

unemployment_clean = unemployment_clean.rename(
    columns={"Year": "year"}
)

# Keep only the fields needed for the analysis.
unemployment_county = unemployment_clean[
    [
        "county_lhj",
        "year",
        "Labor Force",
        "Employment",
        "Unemployment",
        "Unemployment Rate"
    ]
].copy()

# Check the prepared datasets.

# VISUALIZATION 1
# Stakeholder: California state policymakers and economic planners
# Purpose: Show the statewide unemployment trend over time.

statewide_trend = (
    combined_data
    .groupby("year")
    .agg({
        "Labor Force": "sum",
        "Unemployment": "sum"
    })
    .reset_index()
)

# Calculate a weighted statewide unemployment rate.
statewide_trend["Statewide Unemployment Rate"] = (
    statewide_trend["Unemployment"] /
    statewide_trend["Labor Force"] * 100
)

plt.figure(figsize=(12, 6))

plt.plot(
    statewide_trend["year"],
    statewide_trend["Statewide Unemployment Rate"],
    marker="o"
)

plt.title(
    "California Unemployment Rate Trend, 2000–2025",
    fontsize=16
)
plt.xlabel("Year")
plt.ylabel("Unemployment Rate (%)")
plt.grid(alpha=0.3)
plt.xticks(statewide_trend["year"], rotation=45)

plt.tight_layout()
plt.show()


# VISUALIZATION 2
# Stakeholder: County and local workforce development officials
# Purpose: Identify counties with the highest unemployment rates
# in the most recent year and show their population sizes.

latest_year = combined_data["year"].max()

latest_data = combined_data[
    combined_data["year"] == latest_year
].copy()

top_unemployment = (
    latest_data
    .sort_values("Unemployment Rate", ascending=False)
    .head(10)
    .sort_values("Unemployment Rate")
)

plt.figure(figsize=(11, 7))

bars = plt.barh(
    top_unemployment["county_lhj"],
    top_unemployment["Unemployment Rate"]
)

plt.title(
    f"California Counties With the Highest Unemployment Rates, {latest_year}",
    fontsize=15
)
plt.xlabel("Unemployment Rate (%)")
plt.ylabel("County")

# Add unemployment rate and population information to each bar.
for bar, rate, population in zip(
    bars,
    top_unemployment["Unemployment Rate"],
    top_unemployment["population"]
):
    plt.text(
        bar.get_width() + 0.1,
        bar.get_y() + bar.get_height() / 2,
        f"{rate:.1f}% | Pop. {population:,.0f}",
        va="center",
        fontsize=9
    )

plt.xlim(
    0,
    top_unemployment["Unemployment Rate"].max() + 3
)

plt.grid(axis="x", alpha=0.3)
plt.tight_layout()
plt.show()

Key Insights

The integrated data shows that California's unemployment rate changed substantially between 2000 and 2025. The statewide trend shows increased unemployment during the early 2000s, a major rise during the Great Recession that peaked around 2010, and another sharp increase in 2020. Unemployment then declined considerably before beginning to rise again in the most recent years.

At the county level, unemployment conditions vary considerably. In 2025, Imperial County had the highest unemployment rate among California counties, followed by Colusa and Tulare Counties. Population provides additional context because counties with similar unemployment rates may represent very different numbers of residents. Together, the visualizations provide both a broad view of California's labor market and a more detailed view of the counties experiencing the greatest unemployment challenges.
