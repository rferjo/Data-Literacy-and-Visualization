# To Do: Convert numeric columns to proper formats (removing commas and extra spaces)

# Remove extra spaces from all column names
filtered_df.columns = filtered_df.columns.str.strip()

# Clean text columns by removing leading and trailing spaces
text_columns = filtered_df.select_dtypes(include="object").columns

for column in text_columns:
    filtered_df[column] = filtered_df[column].str.strip()

# Attempt to convert text columns containing mostly numeric values
for column in filtered_df.columns:
    if filtered_df[column].dtype == "object" and column != agency_column:
        
        # Remove commas, dollar signs, percent signs, and extra spaces
        cleaned_values = (
            filtered_df[column]
            .astype(str)
            .str.replace(",", "", regex=False)
            .str.replace("$", "", regex=False)
            .str.replace("%", "", regex=False)
            .str.strip()
        )
        
        # Convert values to numbers when possible
        numeric_values = pd.to_numeric(cleaned_values, errors="coerce")
        
        # Convert the column only when most nonmissing values are numeric
        original_nonmissing = filtered_df[column].notna().sum()
        converted_nonmissing = numeric_values.notna().sum()
        
        if original_nonmissing > 0:
            numeric_ratio = converted_nonmissing / original_nonmissing
            
            if numeric_ratio >= 0.75:
                filtered_df[column] = numeric_values
