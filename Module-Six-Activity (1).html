Multi-level Data Exploration Overview¶

Multi-level data exploration refers to the process of examining and analyzing data at different levels of detail or abstraction to uncover patterns, insights, and relationships. It’s like zooming in and out of your data to see both the big picture and the finer details from high-level to low-level exploration.

High-Level Exploration – Look at broad summaries:¶

  • Mean, median, min, max
  • istograms, box plots
  • Count of categories, missing data

Mid-Level Exploration – Compare groups or variables:¶

  • Correlations between variables
  • Group-by aggregations (e.g., average sales per region)
  • Visual comparisons like bar plots or scatter plots

Low-Level Exploration – Dive deep into specific cases:¶

  • Investigating outliers
  • Looking at individual records
  • Tracing data transformations

It’s often used in exploration data analysis (EDA), business intelligence dashboards, or machine learning prep, helping analysts or data scientists move between summaries and specifics to better understand the data.

def create_correlation_heatmap(data):
    import os
    
    print("Aggregating total deaths by country...")
    
    raw_data = pd.read_csv("data/COVID_total_deaths.csv")
    
    aggregate_columns = [
        "World", "Africa", "Asia", "Europe", "North America", 
        "South America", "Oceania", "European Union",
        "High income", "Upper middle income",
        "Lower middle income", "Low income"
    ]
    
    country_columns = [
        column for column in raw_data.columns
        if column != "date" and column not in aggregate_columns
    ]

    print("Selecting top 10 countries by total deaths...")

    latest_totals = (
        raw_data[country_columns]
        .iloc[-1]
        .sort_values(ascending=False)
    )
    
    top_10_countries = latest_totals.head(10).index.tolist()
    
    print("Top 10 countries:")
    print(top_10_countries)
    
    print("Filtering the dataset for the top countries...")

    top_country_data = raw_data[top_10_countries].copy()
    
    print("Preparing country data for correlation...")

    top_country_data = top_country_data.apply(
        pd.to_numeric,
        errors="coerce"
    )
    
    top_country_data = top_country_data.ffill().fillna(0)
    
    print("Computing the correlation matrix...")

    correlation_matrix = top_country_data.corr()
    
    print("Correlation matrix shape:", correlation_matrix.shape)
    display(correlation_matrix.round(2))

    print("Plotting the heatmap...")
    
    plt.figure(figsize=(12, 9))
    
    sns.heatmap(
        correlation_matrix,
        annot=True,
        fmt=".2f",
        cmap="vlag",
        center=0,
        linewidths=0.5,
        square=True,
        cbar_kws={"label": "Correlation Coefficient"}
    )
    
    plt.title(
        "Correlation of COVID-19 Death Trends Among the Top 10 Countries",
        fontsize=14,
        pad=18
    )
    plt.xlabel("Country", fontsize=11)
    plt.ylabel("Country", fontsize=11)
    
    plt.xticks(rotation=45, ha="right")
    plt.yticks(rotation=0)
    plt.tight_layout()
    
    os.makedirs("output", exist_ok=True)
    
    output_path = "output/correlation_heatmap.png"
    plt.savefig(output_path, dpi=300, bbox_inches="tight")
    plt.close()
    
    print(f"Heatmap saved to {output_path}")
    
    display(Image(filename=output_path))

def create_time_series_chart(data):
    import os

    print("Creating global time series chart...")

    raw_data = pd.read_csv("data/COVID_total_deaths.csv")
    raw_data["date"] = pd.to_datetime(raw_data["date"])

    global_deaths = raw_data[["date", "World"]].copy()

    plt.figure(figsize=(12, 6))

    plt.plot(
        global_deaths["date"],
        global_deaths["World"],
        linewidth=2
    )

    peak_index = global_deaths["World"].idxmax()
    peak_date = global_deaths.loc[peak_index, "date"]
    peak_value = global_deaths.loc[peak_index, "World"]

    plt.annotate(
        f"Highest total: {peak_value:,.0f}",
        xy=(peak_date, peak_value),
        xytext=(-160, -45),
        textcoords="offset points",
        arrowprops={"arrowstyle": "->"},
        fontsize=10
    )

    plt.title(
        "Global COVID-19 Deaths Over Time",
        fontsize=14,
        pad=15
    )
    plt.xlabel("Date")
    plt.ylabel("Cumulative Total Deaths")
    plt.grid(True, linestyle="--", alpha=0.4)
    plt.xticks(rotation=45)
    plt.tight_layout()

    os.makedirs("output", exist_ok=True)

    output_path = "output/global_deaths_time_series.png"
    plt.savefig(output_path, dpi=300, bbox_inches="tight")
    plt.close()

    print(f"Time-series chart saved to {output_path}")
    display(Image(filename=output_path))

def clean_data(data):
    print("\nCleaning the dataset...")

    cleaned_data = data.copy()

    # Separate the date column from the numerical columns.
    numeric_columns = cleaned_data.columns.drop("date")

    # Convert all country and region columns to numerical values.
    cleaned_data[numeric_columns] = cleaned_data[numeric_columns].apply(
        pd.to_numeric,
        errors="coerce"
    )

    # Replace missing values with zero.
    cleaned_data[numeric_columns] = cleaned_data[numeric_columns].fillna(0)

    # Remove countries or regions whose values are zero for every date.
    nonzero_columns = numeric_columns[
        (cleaned_data[numeric_columns] != 0).any(axis=0)
    ]

    cleaned_data = cleaned_data[
        ["date"] + nonzero_columns.tolist()
    ]

    # Remove dates where every numerical value is zero.
    cleaned_data = cleaned_data.loc[
        (cleaned_data[nonzero_columns] != 0).any(axis=1)
    ].copy()

    # Convert the date column to datetime.
    cleaned_data["date"] = pd.to_datetime(cleaned_data["date"])

    cleaned_data = cleaned_data.sort_values("date").reset_index(drop=True)

    print("Filtered dataset shape:", cleaned_data.shape)
    display(cleaned_data.head())

    return cleaned_data

def select_top_countries(data):
    print("\nSelecting top 10 countries by total deaths...")

    aggregate_columns = [
        "World",
        "Africa",
        "Europe",
        "North America",
        "South America",
        "Asia",
        "Oceania",
        "European Union (27)",
        "High-income countries",
        "Upper-middle-income countries",
        "Lower-middle-income countries",
        "Low-income countries"
    ]

    country_columns = [
        column for column in data.columns
        if column != "date" and column not in aggregate_columns
    ]

    # Use the most recent cumulative value for each country.
    latest_totals = (
        data[country_columns]
        .iloc[-1]
        .sort_values(ascending=False)
    )

    top_10_countries = latest_totals.head(10).index.tolist()

    print("\nTop 10 individual countries by total deaths:")
    for position, country in enumerate(top_10_countries, start=1):
        print(
            f"{position}. {country}: "
            f"{latest_totals[country]:,.0f}"
        )

    # Keep the date and selected country columns.
    filtered_data = data[
        ["date"] + top_10_countries
    ].copy()

    # Downsample to weekly observations to reduce animation size.
    weekly_data = (
        filtered_data
        .set_index("date")
        .resample("W")
        .max()
        .reset_index()
    )

    print(
        f"\nFiltered dataset shape after selecting "
        f"and resampling: {weekly_data.shape}"
    )

    return weekly_data, top_10_countries
